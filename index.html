/* * DCK ICT — ict-aanwezig (Mijn ICT-Omgeving) * Laat zien wie er nu in Mijn ICT-Omgeving werkt en wie welk ticket open heeft ("oogje"). * De pagina stuurt elke ~20 seconden (op een ander tabblad elke ~40 seconden) een seintje: * { nr: "12601" | null, naam: "…", bezig: true|false, zicht: true|false, stil: true|false (niet storen) } of bij afsluiten { uit: true } * met het AFAS-token van de gebruiker in de Authorization-header. Zonder sleutel daarin valt de functie terug op geheugen. */ const crypto = require('crypto'); const ORIGIN = process.env.MIJNICT_ORIGIN || 'https://rvdveen88.github.io'; const OMGEVING = process.env.AFAS_OMGEVING || '88406'; const CONN_IK = process.env.MIJNICT_CONN_IK || 'PowerBI_ICT_Gebruikers'; const TTL = 90 * 1000, TABEL = 'aanwezig', GEB_TTL = 12 * 3600 * 1000, ACT_SOORTEN = ['gesloten', 'doorgezet', 'opgepakt', 'reactie', 'vermeld', 'gewijzigd']; /* ---------- Table Storage via REST (geen npm-pakketten nodig) ---------- */ const cs = Object.fromEntries(String(process.env.AzureWebJobsStorage || '').split(';').filter(Boolean).map(d => { const i = d.indexOf('='); return [d.slice(0, i), d.slice(i + 1)]; })); const ACCOUNT = cs.AccountName, KEY = cs.AccountKey, SUFFIX = cs.EndpointSuffix || 'core.windows.net'; const heeftOpslag = !!(ACCOUNT && KEY); async function tabel(methode, pad, body, extra = {}) { const datum = new Date().toUTCString(); const resource = `/${ACCOUNT}/${pad.split('?')[0]}`; const sig = crypto.createHmac('sha256', Buffer.from(KEY, 'base64')).update(`${datum}\n${resource}`, 'utf8').digest('base64'); const r = await fetch(`https://${ACCOUNT}.table.${SUFFIX}/${pad}`, { method: methode, headers: { 'x-ms-date': datum, 'x-ms-version': '2019-02-02', 'Authorization': `SharedKeyLite ${ACCOUNT}:${sig}`, 'Accept': 'application/json;odata=nometadata', 'DataServiceVersion': '3.0;NetFx', 'MaxDataServiceVersion': '3.0;NetFx', ...(body ? { 'Content-Type': 'application/json' } : {}), ...extra }, body: body ? JSON.stringify(body) : undefined }); return r; } let tabelKlaar = false; async function zorgTabel() { if (tabelKlaar) return; const r = await tabel('POST', 'Tables', { TableName: TABEL }); if (r.ok || r.status === 409) tabelKlaar = true; } const sleutel = s => encodeURIComponent(String(s).replace(/[\/\\#?']/g, '_')); const geheugen = new Map(); // terugval zonder opslagsleutel: rij -> entiteit async function zet(ent) { if (!heeftOpslag) { geheugen.set(ent.RowKey, ent); return; } await zorgTabel(); await tabel('PUT', `${TABEL}(PartitionKey='p',RowKey='${sleutel(ent.RowKey)}')`, ent); } async function weg(rk) { if (!heeftOpslag) { geheugen.delete(rk); return; } await tabel('DELETE', `${TABEL}(PartitionKey='p',RowKey='${sleutel(rk)}')`, null, { 'If-Match': '*' }); } async function actueel() { const grens = Date.now() - TTL; if (!heeftOpslag) { for (const [k, v] of geheugen) if (v.t < grens) geheugen.delete(k); return [...geheugen.values()]; } await zorgTabel(); const r = await tabel('GET', `${TABEL}()?$filter=${encodeURIComponent(`PartitionKey eq 'p' and t gt ${grens}L`)}`); if (!r.ok) return []; return ((await r.json()).value || []); } /* ---------- gebeurtenissen (kritieke storing) ---------- */ const geheugenGeb = []; async function zetGebeurtenis(ent) { if (!heeftOpslag) { geheugenGeb.push(ent); return; } await zorgTabel(); await tabel('PUT', `${TABEL}(PartitionKey='g',RowKey='${sleutel(ent.RowKey)}')`, ent); // oude gebeurtenissen opruimen const r = await tabel('GET', `${TABEL}()?$filter=${encodeURIComponent(`PartitionKey eq 'g' and t lt ${Date.now() - GEB_TTL}L`)}`); if (r.ok) for (const e of ((await r.json()).value || [])) await tabel('DELETE', `${TABEL}(PartitionKey='g',RowKey='${sleutel(e.RowKey)}')`, null, { 'If-Match': '*' }).catch(() => {}); } async function gebeurtenissen() { const grens = Date.now() - GEB_TTL; if (!heeftOpslag) { while (geheugenGeb.length && geheugenGeb[0].t < grens) geheugenGeb.shift(); return [...geheugenGeb]; } const r = await tabel('GET', `${TABEL}()?$filter=${encodeURIComponent(`PartitionKey eq 'g' and t gt ${grens}L`)}`); if (!r.ok) return []; return ((await r.json()).value || []); } /* ---------- profielfoto's ---------- */ const geheugenAv = new Map(); async function zetAvatar(rk, ent) { if (!heeftOpslag) { if (ent) geheugenAv.set(rk, ent); else geheugenAv.delete(rk); return; } await zorgTabel(); if (ent) await tabel('PUT', `${TABEL}(PartitionKey='a',RowKey='${sleutel(rk)}')`, { PartitionKey: 'a', RowKey: rk, ...ent }); else await tabel('DELETE', `${TABEL}(PartitionKey='a',RowKey='${sleutel(rk)}')`, null, { 'If-Match': '*' }).catch(() => {}); } async function avatars(metFoto) { if (!heeftOpslag) return [...geheugenAv.values()].map(a => ({ code: a.code, naam: a.naam, t: a.t, ...(metFoto ? { d: a.d } : {}) })); const r = await tabel('GET', `${TABEL}()?$filter=${encodeURIComponent("PartitionKey eq 'a'")}${metFoto ? '' : '&$select=code,naam,t'}`); if (!r.ok) return []; return ((await r.json()).value || []).map(a => ({ code: a.code, naam: a.naam, t: a.t, ...(metFoto ? { d: a.d } : {}) })); } /* ---------- token controleren bij AFAS (15 minuten onthouden) ---------- */ const tokens = new Map(); // sha256(token) -> {code, tot} async function wie(token) { const h = crypto.createHash('sha256').update(token).digest('hex'), c = tokens.get(h); if (c && c.tot > Date.now()) return c.code; const r = await fetch(`https://${OMGEVING}.rest.afas.online/ProfitRestServices/connectors/${encodeURIComponent(CONN_IK)}?skip=0&take=1`, { headers: { Authorization: 'Bearer ' + token, Accept: 'application/json' } }); if (!r.ok) return null; const rij = ((await r.json()).rows || [])[0] || {}; const code = String(rij.Gebruiker || rij.User || rij.Gebruikerscode || Object.values(rij)[0] || 'onbekend'); tokens.set(h, { code, tot: Date.now() + 15 * 60 * 1000 }); if (tokens.size > 500) for (const [k, v] of tokens) if (v.tot < Date.now()) tokens.delete(k); return code; } module.exports = async function (context, req) { const cors = { 'Content-Type': 'application/json', 'Cache-Control': 'no-store', 'Access-Control-Allow-Origin': ORIGIN, 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization' }; const antwoord = (status, body) => { context.res = { status, headers: cors, body: JSON.stringify(body) }; }; if (req.method === 'OPTIONS') { context.res = { status: 204, headers: cors }; return; } const token = String((req.headers && (req.headers.authorization || req.headers.Authorization)) || '').replace(/^Bearer\s+/i, ''); if (!token) { antwoord(401, { error: 'geen token' }); return; } try { const code = await wie(token); if (!code) { antwoord(401, { error: 'token ongeldig' }); return; } const b = req.body || {}, nr = b.nr ? String(b.nr).slice(0, 20) : ''; const rk = sleutel(code); // één rij per gebruiker: online of niet, met hooguit één open ticket if (b.uit) { await weg(rk).catch(() => {}); antwoord(200, { ik: code, lijst: [] }); return; } const g = b.gebeurtenis; if (g && g.soort === 'kritiek' && g.nr) { const t = Date.now(); await zetGebeurtenis({ PartitionKey: 'g', RowKey: `${t}-${rk}`, soort: 'kritiek', nr: String(g.nr).slice(0, 20), onderwerp: String(g.onderwerp || '').slice(0, 160), toelichting: String(g.toelichting || '').slice(0, 400), door: String(b.naam || code).slice(0, 80), code, sessie: String(g.sessie || '').slice(0, 20), t }); } if (g && ACT_SOORTEN.includes(g.soort) && g.nr) { const t = Date.now(); await zetGebeurtenis({ PartitionKey: 'g', RowKey: `${t}-${rk}-${String(g.nr).slice(0, 20)}`, soort: g.soort, nr: String(g.nr).slice(0, 20), onderwerp: String(g.onderwerp || '').slice(0, 160), naar: String(g.naar || '').slice(0, 200), tekst: String(g.tekst || '').slice(0, 200), w: typeof g.w === 'string' ? g.w.slice(0, 1500) : '', door: String(b.naam || code).slice(0, 80), code, sessie: String(g.sessie || '').slice(0, 20), t }); } let avatarOk = false; if (typeof b.avatar === 'string') { if (b.avatar === '') { await zetAvatar(rk, null); avatarOk = true; } else if (b.avatar.length <= 40000 && /^data:image\/(jpeg|png|webp);base64,[A-Za-z0-9+/=]+$/.test(b.avatar)) { await zetAvatar(rk, { code, naam: String(b.naam || code).slice(0, 80), d: b.avatar, t: Date.now() }); avatarOk = true; } } await zet({ PartitionKey: 'p', RowKey: rk, nr, code, naam: String(b.naam || code).slice(0, 80), bezig: !!(nr && b.bezig), typt: !!(nr && b.typt), zicht: b.zicht !== false, stil: !!b.stil, t: Date.now() }); const lijst = (await actueel()).filter(e => e.t >= Date.now() - TTL).map(e => ({ nr: e.nr || '', code: e.code, naam: e.naam, bezig: !!e.bezig, typt: !!e.typt, zicht: e.zicht !== false, stil: !!e.stil, t: e.t })); const alleGeb = (await gebeurtenissen()).sort((a, b) => a.t - b.t), act = alleGeb.filter(e => e.soort !== 'kritiek').slice(-150); const geb = [...alleGeb.filter(e => e.soort === 'kritiek'), ...act] .map(e => ({ id: e.RowKey, soort: e.soort, nr: e.nr, onderwerp: e.onderwerp, toelichting: e.toelichting, naar: e.naar || '', tekst: e.tekst || '', w: e.w || '', door: e.door, code: e.code, sessie: e.sessie || '', t: e.t })); const av = await avatars(!!b.avatars).catch(() => []); antwoord(200, { ik: code, lijst, gebeurtenissen: geb, ...(b.avatars ? { avatars: av } : { av }), ...(avatarOk ? { avatarOk } : {}), opslag: heeftOpslag ? 'tabel' : 'geheugen' }); } catch (err) { context.log('ict-aanwezig fout:', err); antwoord(500, { error: err.message }); } };
